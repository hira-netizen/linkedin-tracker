function doPost(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getActiveSheet();
    var data = JSON.parse(e.postData.contents || '{}');

    setupOrUpdateSheet(sheet);

    var rowNum = findExistingLeadRow(sheet, data);
    if (!rowNum) rowNum = sheet.getLastRow() + 1;

    var nextReminder = getNextReminder(data);

    writeByHeader(sheet, rowNum, {
      'Name': data.name || '',
      'Job Title': data.jobTitle || '',
      'Niche / Expertise': data.niche || '',
      'Source': data.source || 'LinkedIn',
      'Profile URL': data.url || '',
      'Status': data.status || 'Lead Added',
      'Date of Engagement': data.dateEngaged || '',
      'Follow Up 1': data.followUp1 || '',
      'Final Follow Up': data.finalFollowUp || '',
      'Outcome': data.outcome || '',
      'DM Message': data.dm || '',
      'Notes': data.notes || '',
      'Last Updated': new Date(),
      'Next Reminder': nextReminder
    });

    formatRow(sheet, rowNum, data.source || 'LinkedIn');
    formatWholeSheet(sheet);
    applyDropdowns(sheet);

    return ContentService.createTextOutput(JSON.stringify({success:true,row:rowNum})).setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({success:false,error:err.toString()})).setMimeType(ContentService.MimeType.JSON);
  }
}

function setupOrUpdateSheet(sheet) {
  var headers = ['Name','Job Title','Niche / Expertise','Source','Profile URL','Status','Date of Engagement','Follow Up 1','Final Follow Up','Outcome','DM Message','Notes','Last Updated','Next Reminder'];

  if (sheet.getLastRow() === 0) {
    sheet.appendRow(headers);
  } else {
    var existing = sheet.getRange(1,1,1,Math.max(sheet.getLastColumn(),1)).getValues()[0];
    headers.forEach(function(h) {
      if (existing.indexOf(h) === -1) {
        sheet.getRange(1, sheet.getLastColumn()+1).setValue(h);
        existing.push(h);
      }
    });
  }

  var hr = sheet.getRange(1,1,1,sheet.getLastColumn());
  hr.setBackground('#111827').setFontColor('#ffffff')
    .setFontWeight('bold').setFontSize(11)
    .setHorizontalAlignment('center');
  sheet.setFrozenRows(1);
  setColumnWidths(sheet);
  applyDropdowns(sheet);
}

function headerMap(sheet) {
  var headers = sheet.getRange(1,1,1,sheet.getLastColumn()).getValues()[0];
  var map = {};
  headers.forEach(function(h,i){ map[h] = i+1; });
  return map;
}

function writeByHeader(sheet, row, values) {
  var map = headerMap(sheet);
  Object.keys(values).forEach(function(key){
    if (map[key]) sheet.getRange(row, map[key]).setValue(values[key]);
  });
}
