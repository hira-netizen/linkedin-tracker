function doPost(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getActiveSheet();
    var data = JSON.parse(e.postData.contents);

    // Setup header row if empty
    if (sheet.getLastRow() === 0) setupSheet(sheet);

    // Duplicate check by name (column A)
    var lastRow = sheet.getLastRow();
    if (lastRow > 1) {
      var names = sheet.getRange(2, 1, lastRow - 1, 1).getValues();
      for (var i = 0; i < names.length; i++) {
        if (names[i][0].toString().toLowerCase().trim() === data.name.toLowerCase().trim()) {
          return ContentService
            .createTextOutput(JSON.stringify({success:false, duplicate:true}))
            .setMimeType(ContentService.MimeType.JSON);
        }
      }
    }

    // Append new row
    var newRow = lastRow + 1;
    sheet.appendRow([
      data.name || '',
      data.dm || '',
      data.source || 'LinkedIn',
      data.niche || '',
      data.engagement || 'yes',
      data.replyStatus || '',
      data.dateOfMessage || '',
      data.followUp1 || '',
      data.finalFollowUp || '',
      data.notes || ''
    ]);

    // Style the new row based on platform
    formatRow(sheet, newRow, data.source);

    return ContentService
      .createTextOutput(JSON.stringify({success:true}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService
      .createTextOutput(JSON.stringify({success:false, error:err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function setupSheet(sheet) {
  // Headers
  var headers = ['Name','DM Message','Source','Niche/Expertise',
                 'Engagement','Reply Status','Date of Message',
                 'Follow Up 1','Final Follow Up','Notes'];
  sheet.appendRow(headers);

  // Style header row
  var headerRange = sheet.getRange(1, 1, 1, 10);
  headerRange.setBackground('#1a1a2e');
  headerRange.setFontColor('#ffffff');
  headerRange.setFontWeight('bold');
  headerRange.setFontSize(11);
  headerRange.setHorizontalAlignment('center');

  // Column widths
  sheet.setColumnWidth(1, 160);  // Name
  sheet.setColumnWidth(2, 280);  // DM
  sheet.setColumnWidth(3, 110);  // Source
  sheet.setColumnWidth(4, 180);  // Niche
  sheet.setColumnWidth(5, 100);  // Engagement
  sheet.setColumnWidth(6, 130);  // Reply Status
  sheet.setColumnWidth(7, 130);  // Date
  sheet.setColumnWidth(8, 130);  // Follow Up 1
  sheet.setColumnWidth(9, 130);  // Final Follow Up
  sheet.setColumnWidth(10, 250); // Notes

  // Freeze header row
  sheet.setFrozenRows(1);
}

function formatRow(sheet, rowNum, source) {
  var range = sheet.getRange(rowNum, 1, 1, 10);

  // Platform colours
  var bgColor = '#ffffff';
  var accentColor = '#cccccc';
  if (source === 'LinkedIn')  { bgColor = '#e8f4fb'; accentColor = '#0077b5'; }
  if (source === 'Instagram') { bgColor = '#fce4ec'; accentColor = '#e1306c'; }
  if (source === 'Facebook')  { bgColor = '#ede9fe'; accentColor = '#7c3aed'; }

  // Row background
  range.setBackground(bgColor);
  range.setFontSize(10);
  range.setVerticalAlignment('middle');

  // Bold the name
  sheet.getRange(rowNum, 1).setFontWeight('bold').setFontSize(11);

  // Colour the source cell with platform colour
  sheet.getRange(rowNum, 3)
    .setBackground(accentColor)
    .setFontColor('#ffffff')
    .setFontWeight('bold')
    .setHorizontalAlignment('center');

  // Add borders
  range.setBorder(true, true, true, true, false, false,
    '#cccccc', SpreadsheetApp.BorderStyle.SOLID);

  // Alternate row shading (slightly darker every other row)
  if (rowNum % 2 === 0) {
    range.setBackground(blendColor(bgColor));
  }

  // Wrap text for DM and Notes columns
  sheet.getRange(rowNum, 2).setWrap(true);
  sheet.getRange(rowNum, 10).setWrap(true);
}

function blendColor(hex) {
  // Makes colour ~10% darker for alternating rows
  var r = parseInt(hex.slice(1,3),16);
  var g = parseInt(hex.slice(3,5),16);
  var b = parseInt(hex.slice(5,7),16);
  r = Math.max(0, r-15); g = Math.max(0, g-15); b = Math.max(0, b-15);
  return '#' + [r,g,b].map(x=>x.toString(16).padStart(2,'0')).join('');
}
